kemudian, izinkan port koneksi dengan update rules firewall dengan perintah

ufw = uncomplecated firewall

sudo unf allow OpenSSH -> ini port dari ssh 22/tcp

untuk membuka port web
membuka web http  = sudo ufw allow http  -> port 80
membuka web https = sudo ufw allow https -> port 443

kemudian buka port database (contoh mysql) <opsional>
sudo ufw allow mysql

setelah rules di update di firewall "enable" setiap portnya dengan

sudo ufw enable

lalu y (yes)

lalu cek dengan | sudo ufw status

sudo nano /etc/nginx/sites-available/ecommerce

dan isi:

server {
    listen 80;
    listen [::]:80;

    server_name ecommercekudadiri.my.id www.ecommercekudadiri.my.id;

    # =========================
    # React Frontend
    # =========================
    root /var/www/ecommercekudadiri.my.id/ecommerce-laravel/frontend_ecommerce/dist;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # =========================
    # Laravel Storage
    # =========================
    location /storage/ {
        alias /var/www/ecommercekudadiri.my.id/ecommerce-laravel/backend_ecommerce/public/storage/;
        access_log off;
        expires 30d;
    }

    # =========================
    # Laravel API
    # =========================
    # location /api {
    #     root /var/www/ecommercekudadiri.my.id/ecommerce-laravel/backend_ecommerce/public;
    #     index index.php;
    #     try_files $uri /index.php?$query_string;
    # }
	
    location /api {
        root /var/www/ecommercekudadiri.my.id/ecommerce-laravel/backend_ecommerce/public;
        try_files /index.php?$query_string;
    }

    location ~ \.php$ {
        root /var/www/ecommercekudadiri.my.id/ecommerce-laravel/backend_ecommerce/public;

        include snippets/fastcgi-php.conf;
        fastcgi_pass unix:/run/php/php8.3-fpm.sock;

        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
    }

    location ~ /\. {
        deny all;
    }

    client_max_body_size 20M;
}
